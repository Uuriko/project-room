const wait = async (predicate, label) => {
  const deadline = Date.now() + 20000;
  while (Date.now() < deadline) { if (predicate()) return; await new Promise(resolve => setTimeout(resolve, 25)); }
  throw new Error('Timed out: ' + label);
};
