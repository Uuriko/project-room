import {NormalizedExportRoom} from './normalized-owner-export.test-fixture.mjs';
import {withCurrentReadContext} from '../server/current-room-repository.mjs';
export class ExportBudgetRoom extends NormalizedExportRoom {
 async fetch(request){
  if(new URL(request.url).pathname==='/__budget'){
   const token=request.headers.get('Authorization')?.slice(7),results=[];
   for(const limit of [16,8,4,1])this.store.readTransaction(()=>withCurrentReadContext(this.store.db,'native-export',b=>{
    try{const page=this.store.eventsAfter(token,'native-export',0,limit);results.push({limit,ok:true,shown:page.events.length,more:page.hasMore,bytes:b.bytes,records:b.records});}
    catch(e){results.push({limit,ok:false,code:e.code,message:e.message,bytes:b.bytes,records:b.records});}
   }));
   return Response.json(results);
  }
  return super.fetch(request);
 }
}
export {default} from './normalized-owner-export.test-fixture.mjs';
